# ── Dual-GPU Training (NCCL, no DDP) ─────────────────────────────────
# Two model replicas process different micro-batches in parallel.
# Gradients flattened + allreduced via NCCL AFTER backward (no DDP).
# Effective batch size = 2 (1 per GPU).
#
# Throughput: ~0.8k tok/s (2× single GPU)
#
# To stop:
# pkill -9 -f "venv_nv/bin/python" && echo "Killed training processes" || echo "No training process found"
# fuser -k -9 29501/tcp && echo "Freed PyTorch distributed port" || echo "Port already free"


cd /home/kyler/Documents/CTM-Transformer && \
  source /home/kyler/Documents/OpenMythos/venv_nv/bin/activate && \
  CUDA_VISIBLE_DEVICES=0,1 PYTHONUNBUFFERED=1 python \
    -m ctm_transformer.train_offload \
    --device cuda \
    --dtype bfloat16 \
    --d_model 1024 \
    --d_latent 1024 \
    --n_layers 24 \
    --n_heads 16 \
    --nlm_hidden_dim 64 \
    --nlm_groups 1 \
    --history_len 8 \
    --max_thought_steps 8 \
    --use_hyperloop \
    --hyperloop_n_begin 4 \
    --hyperloop_n_middle 18 \
    --hyperloop_n_end 4 \
    --hyperloop_middle_loops 1 \
    --seq_len 512 \
    --batch_size 1 \
    --gradient_accumulation_steps 8 \
    --learning_rate 3e-4 \
    --warmup_steps 2000 \
    --max_steps 200000 \
    --no_gradient_checkpointing \
    --log_interval 10 \
    --eval_interval 100000 \
    --optimizer adamw \
    --use_8bit_adam \
    --gradient_checkpointing_min_T 2 \
    --checkpoint_dir checkpoints_v2_dual \
    --use_feec \
    --use_matrix_streams \
    --n_streams 4 \
    --use_loop_pos_emb \
    --use_triton_attention \
    --tiled_schedule \
    --sync_method sparse_decay \
    --sync_sparse_pairs 256 \
    --synapse_type mlp \
    --temporal_loss_type dynamic_aggregate \
    --use_two_phase_curriculum \
    --use_distillation \
    --use_cached_teacher \
    --teacher_cache_dir ./teacher_cache \
    --tokenizer "hf:nvidia/NVIDIA-Nemotron-3-Nano-4B-BF16" \
    --phase1_tokens 500000000 \
    --distill_logit_weight 1.0 \
    --distill_temperature 4.0 \
    --distill_tick_aggregation first \
    --distill_top_k 128 \
    --distill_feature_weight 0.0 \
    --use_shared_head_film \
    --tie_embeddings \
    > train_dual_gpu_teacher_dynamicaggregate.log 2>&1 &
echo "Dual-GPU Training PID: $!"
