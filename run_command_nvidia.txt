#   LAUNCH:
#     `torchrun --standalone --nproc_per_node=2 -m ctm_transformer.train`
#     instead of `python -m ctm_transformer.train`. torchrun spawns 2
#     processes (one per GPU), sets RANK / LOCAL_RANK / WORLD_SIZE in
#     env, and the train script reads those to set up DDP.
#
#   ADDED FLAGS:
#     (none — DDP is auto-detected from torchrun env vars)
#
#   AUTO-DISABLED UNDER DDP:
#     --use_cuda_graphs (CUDA Graphs and DDP NCCL all-reduce don't

#   NEW STARTUP LOG LINES:
#     [DDP] Wrapped model. world_size=2, this rank=0, local_rank=0
#
#   THROUGHPUT METRIC:
#     `tok/s` in the log line is now the EFFECTIVE rate across both
#     ranks (per-rank tokens × world_size). Should jump from ~400
#     (single-GPU) to ~750-900 (two-GPU DDP without NVLink).

cd /home/kyler/Documents/CTM-Transformer && source /home/kyler/Documents/OpenMythos/venv_nv/bin/activate && CUDA_VISIBLE_DEVICES=0,1 PYTHONUNBUFFERED=1 nohup torchrun \
  --standalone --nproc_per_node=2 \
  -m ctm_transformer.train \
  --device cuda \
  --dtype bfloat16 \
  --d_model 1024 \
  --d_latent 1024 \
  --n_layers 24 \
  --n_heads 16 \
  --nlm_hidden_dim 64 \
  --nlm_groups 1 \
  --history_len 8 \
  --max_thought_steps 8 \
  --use_hyperloop \
  --hyperloop_n_begin 4 \
  --hyperloop_n_middle 18 \
  --hyperloop_n_end 4 \
  --hyperloop_middle_loops 1 \
  --seq_len 512 \
  --batch_size 1 \
  --gradient_accumulation_steps 1 \
  --learning_rate 3e-4 \
  --warmup_steps 2000 \
  --max_steps 10000000 \
  --gradient_checkpointing \
  --log_interval 10 \
  --eval_interval 1000 \
  --optimizer adamw \
  --use_8bit_adam \
  --gradient_checkpointing_min_T 2 \
  --checkpoint_dir checkpoints_v2_ddp \
  --use_feec \
  --n_streams 4 \
  --use_loop_pos_emb \
  --use_triton_attention \
  --tiled_schedule \
  --sync_method sparse_decay \
  --sync_sparse_pairs 256 \
  --synapse_type mlp \
  --temporal_loss_type dynamic_aggregate \
  --use_two_phase_curriculum \
  --use_distillation \
  --use_cached_teacher \
  --teacher_cache_dir ./teacher_cache \
  --tokenizer "hf:nvidia/NVIDIA-Nemotron-3-Nano-4B-BF16" \
  --phase1_tokens 500000000 \
  --distill_logit_weight 1.0 \
  --distill_temperature 4.0 \
  --distill_tick_aggregation all \
  --distill_top_k 128 \
  --distill_feature_weight 0.0 \
  --use_shared_head_film \
  --tie_embeddings \
  --profile \
  --profile_warmup 20 \
  --profile_interval 10000 \
  > train_v2_ddp.log 2>&1 &
echo "DDP PID (torchrun parent): $!"

#      with NCCL_DEBUG=INFO.

#   5. Loss values should be similar to (but not identical to) the
#      single-GPU run. With DDP, each step's gradient is averaged
#      across ranks, so it's mathematically equivalent to a single
#      machine running batch_size=2*batch_size. Convergence
#      trajectory should be similar in shape.

# ─────────────────────────────────────────────────────────────────────
# Diagnosing DDP problems
# ─────────────────────────────────────────────────────────────────────
#
# If the run hangs at startup before any training begins:
#   - Likely NCCL initialization. Add NCCL_DEBUG=INFO to env to see
#     where it's stuck. Typical issues: firewall blocking the rendezvous
#     port, or another process already using port 29500.
#
# If the run starts but hangs after a few backward passes:
#   - find_unused_parameters detection issue. The model has tick-
#     conditional FiLM access — if a tick is ever skipped, that step's
#     unused params will trigger DDP's hang-on-mismatch behavior.
#     Confirm by adding TORCH_DISTRIBUTED_DEBUG=DETAIL to env.
#
# If throughput is BELOW 1.5× single-GPU:
#   - PCIe contention. Check that both GPUs are at full PCIe gen 4 x16:
#     `nvidia-smi --query-gpu=pcie.link.gen.current,pcie.link.width.current --format=csv`
#   - Check if CPU-side dispatch is the bottleneck (rare on 3090): the
#     PhaseTimer report should show student_fwd close to single-GPU value;
#     if it's 2× single-GPU value, something's wrong with the device
#     binding.
# ─────────────────────────────────────────────────────────────────────
